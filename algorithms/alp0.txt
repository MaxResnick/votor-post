 1  state per slot s:
 2      voted(s)          // my one first-round vote
 3      voted_notar(s)    // which block I notarized
 4      bad_window(s)     // I cast Skip or a fallback vote
 5      its_over(s)       // I cast Finalize
 6      parent_ready(s)   // certs justify this parent

 7  upon ParentReady(s, parent):
 8      enter slot s; start local timer T_s
 9      if I am the leader of s:
10          broadcast Propose(s, b, justification)

11  upon Block(s, b):          // b completed replay
12      if voted(s):
13          do nothing
14      else if parent_ready(s, parent(b)):
15          broadcast Notarize(b)
16      else:
17          hold b as pending, retry on new certificates

18  at T_s = 2Δ:
19      if not voted(s):
20          broadcast Skip(s)
21          set bad_window(s)

22  upon receiving 60% Notarize votes for b in slot s:
23      if voted_notar(s) = b and not bad_window(s) and not its_over(s):
24          broadcast Finalize(s, b)
25          set its_over(s)

26  upon receiving 40% Notarize votes for b in slot s, or 20% Notarize for b with 60% Notarize-or-Skip for s:
27      if not voted(s):
28          broadcast Skip(s); set bad_window(s)
29      if not its_over(s) and no NotarizeFallback(b) before:
30          broadcast NotarizeFallback(b)
31          set bad_window(s)

32  upon receiving 40% Skip votes for s, counting Notarize votes for non-leading blocks:
33      if not voted(s):
34          broadcast Skip(s); set bad_window(s)
35      if not its_over(s) and no SkipFallback(s) before:
36          broadcast SkipFallback(s)
37          set bad_window(s)

38  upon receiving 80% Notarize votes for b, or 60% Finalize votes for s plus a notarization cert:
39      finalize s and all its ancestors
