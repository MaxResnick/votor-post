 1  state per slot s:
 2      voted(s)          // my one first-round vote
 3      voted_notar(s)    // which block I notarized
 4      bad_window(s)     // I cast Skip or SkipFallback
 5      its_over(s)       // I cast Finalize
 6      parent_ready(s)   // certs justify this parent

 7  upon ParentReady(s, parent):
 8      enter slot s; start local timer T_s
 9      if I am the leader of s:
10          broadcast Propose(s, b, justification)

11  upon Block(s, b):          // b completed replay
12      if voted(s):
13          do nothing
14      else if parent_ready(s, parent(b)) and T_s &lt; 2Δ:
15          broadcast Notarize(b)
16      else:
17          hold b as pending, retry on new certificates

18  at T_s = 2Δ:
19      if not voted(s):
20          broadcast Skip(s)
21          set bad_window(s)

22  upon receiving <mark>67%</mark> Notarize votes for b in slot s:
23      if not bad_window(s) and not its_over(s):
24          broadcast Finalize(s, b)
25          set its_over(s)

26  <mark class="add">at T_s = 3Δ:</mark>
27      <mark class="add">if voted_notar(s) and not bad_window(s) and not its_over(s):</mark>
28          <mark class="add">broadcast SkipFallback(s)  // release: slot failed</mark>
29          <mark class="add">set bad_window(s)</mark>

30  upon receiving <mark>67%</mark> Finalize votes for b in slot s:
31      finalize b and all its ancestors
